cmake_minimum_required(VERSION 3.25)
project(neutrino VERSION 0.15.0 LANGUAGES C)

if(CMAKE_C_COMPILER_ID STREQUAL "MSVC")
    message(FATAL_ERROR "Neutrino requires C23. Use a C23-capable GCC or Clang, including clang-cl on Windows.")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
    add_compile_options(/utf-8 "$<$<CONFIG:Release,RelWithDebInfo>:/clang:-O3>")
else()
    add_compile_options("$<$<CONFIG:Release,RelWithDebInfo>:-O3>")
endif()

find_package(CURL 7.62 REQUIRED)
set(browser_header "${CMAKE_CURRENT_BINARY_DIR}/browser_assets.h")
set(browser_assets "")
foreach(asset IN ITEMS index.html app.css app.js)
    set(path "${CMAKE_CURRENT_SOURCE_DIR}/web/${asset}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
    string(MAKE_C_IDENTIFIER "${asset}" identifier)
    file(READ "${path}" data HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," data "${data}")
    string(APPEND browser_assets "static const unsigned char ${identifier}[] = {${data}0};\n")
endforeach()
file(GENERATE OUTPUT "${browser_header}" CONTENT "${browser_assets}")
add_executable(neutrino src/main.c src/workspace.c src/terminal.c src/inference.c src/gateway.c src/memory.c src/facts.c src/world.c src/ui.c src/web.c
    src/simulation.c src/scenario.c src/physics.c src/rigid.c src/rigid_spin.c src/spill.c src/runoff.c
    src/physics_flow.c src/physics_models.c src/mass_transfer.c src/measurements.c src/ray_gpu.c src/prose.c src/prompt.c src/browser.c
    third_party/cjson/cJSON.c)
target_include_directories(neutrino PRIVATE third_party/cjson "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(neutrino PRIVATE CURL::libcurl)
if(MSVC)
    target_compile_options(neutrino PRIVATE /W4)
    target_compile_definitions(neutrino PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(neutrino PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(neutrino PRIVATE winhttp ws2_32)
    if(MINGW)
        target_link_options(neutrino PRIVATE -municode)
    endif()
else()
    find_package(Threads REQUIRED)
    target_compile_definitions(neutrino PRIVATE _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700)
    target_link_libraries(neutrino PRIVATE Threads::Threads m ${CMAKE_DL_LIBS})
endif()
install(TARGETS neutrino)

option(NEUTRINO_VULKAN "Enable optional Vulkan compute for ray batches on Windows and Linux" OFF)
option(NEUTRINO_VULKAN_RT "Enable optional Vulkan hardware ray queries" OFF)
if(NEUTRINO_VULKAN_RT AND NOT NEUTRINO_VULKAN)
    message(FATAL_ERROR "NEUTRINO_VULKAN_RT requires NEUTRINO_VULKAN")
endif()
if(NEUTRINO_VULKAN)
    if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The Vulkan compute backend supports Windows and Linux.")
    endif()
    find_path(NEUTRINO_VULKAN_INCLUDE_DIR vulkan/vulkan.h
        HINTS "$ENV{VULKAN_SDK}" PATH_SUFFIXES Include include REQUIRED)
    set(NEUTRINO_VULKAN_SHADER_HEADER "" CACHE FILEPATH "Precompiled C header for the Vulkan ray shader")
    set(shader_header "${CMAKE_CURRENT_BINARY_DIR}/neutrino_ray_spv.h")
    if(NEUTRINO_VULKAN_SHADER_HEADER)
        configure_file("${NEUTRINO_VULKAN_SHADER_HEADER}" "${shader_header}" COPYONLY)
    else()
        find_program(NEUTRINO_GLSLANG NAMES glslangValidator glslang
            HINTS "$ENV{VULKAN_SDK}" PATH_SUFFIXES Bin bin REQUIRED)
        add_custom_command(OUTPUT "${shader_header}"
            COMMAND "${NEUTRINO_GLSLANG}" -V --target-env vulkan1.0
                --vn neutrino_ray_spv -o "${shader_header}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/trace_rays.comp"
            DEPENDS shaders/trace_rays.comp
            VERBATIM)
        add_custom_target(neutrino_vulkan_shader DEPENDS "${shader_header}")
    endif()
    if(NEUTRINO_VULKAN_RT)
        set(NEUTRINO_VULKAN_RT_SHADER_HEADER "" CACHE FILEPATH "Precompiled C header for the Vulkan ray query shader")
        set(rt_header "${CMAKE_CURRENT_BINARY_DIR}/neutrino_ray_query_spv.h")
        if(NEUTRINO_VULKAN_RT_SHADER_HEADER)
            configure_file("${NEUTRINO_VULKAN_RT_SHADER_HEADER}" "${rt_header}" COPYONLY)
        else()
            find_program(NEUTRINO_GLSLANG NAMES glslangValidator glslang
                HINTS "$ENV{VULKAN_SDK}" PATH_SUFFIXES Bin bin REQUIRED)
            add_custom_command(OUTPUT "${rt_header}"
                COMMAND "${NEUTRINO_GLSLANG}" -V --target-env vulkan1.2
                    --vn neutrino_ray_query_spv -o "${rt_header}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/ray_query.comp"
                DEPENDS shaders/ray_query.comp
                VERBATIM)
            add_custom_target(neutrino_vulkan_rt_shader DEPENDS "${rt_header}")
        endif()
    endif()
    foreach(target IN ITEMS neutrino scene_physics_test ray_scaling_bench)
        if(TARGET "${target}")
            target_sources("${target}" PRIVATE src/ray_vulkan.c)
            target_compile_definitions("${target}" PRIVATE NEUTRINO_VULKAN=1)
            target_include_directories("${target}" PRIVATE "${NEUTRINO_VULKAN_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
            if(TARGET neutrino_vulkan_shader)
                add_dependencies("${target}" neutrino_vulkan_shader)
            endif()
            if(NEUTRINO_VULKAN_RT)
                target_compile_definitions("${target}" PRIVATE NEUTRINO_VULKAN_RT=1)
                if(TARGET neutrino_vulkan_rt_shader)
                    add_dependencies("${target}" neutrino_vulkan_rt_shader)
                endif()
            endif()
        endif()
    endforeach()
endif()
